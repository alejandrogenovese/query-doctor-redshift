-- top 10 clientes por monto en 2026 WITH mov_filtrados AS ( SELECT party_id, SUM(monto) AS total FROM finanzas.movimientos WHERE fecha_movimiento >= DATE '2026-01-01' GROUP BY party_id ) SELECT c.nombre, m.total FROM crm.clientes c JOIN mov_filtrados m ON c.party_id = m.party_id ORDER BY m.total DESC LIMIT 10;
